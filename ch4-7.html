一、逐步觀察 BST 建構

規則：左小右大。數值：{50, 30, 70, 20, 40}

${animSteps[window.bstStep].label}

二、挑戰任務

`; }; window.moveBST = (dir) => { const next = window.bstStep + dir; if(next >= 0 && next < animSteps.length) { window.bstStep = next; window.renderBSTInner(); } }; window.submitBST = () => { let isAllCorrect = true; let missing = false; challengeNodes.forEach(n => { const val = (window.bstAns[n.id] || '').trim(); if(!val) missing = true; if(val !== n.ans) isAllCorrect = false; }); if(missing) { alert("還有節點沒填數字喔！"); return; } if(isAllCorrect) { if (typeof stopTimer === 'function') stopTimer(); if (typeof showResult === 'function') showResult(true); } else { window.bstStatus = 'error'; window.renderBSTInner(); alert("有些數字放錯位置了！請記得：左子節點 < 父節點 < 右子節點。"); } }; window.renderBSTInner(); }
